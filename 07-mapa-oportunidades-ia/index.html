Ver script SQL para crear la tabla en Supabase

Si aún no has creado la tabla opportunities en Supabase, copia y pega esto en el SQL Editor de Supabase:

CREATE TABLE IF NOT EXISTS public.opportunities (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name TEXT NOT NULL,
  city TEXT NOT NULL,
  country TEXT NOT NULL,
  problem TEXT NOT NULL,
  ai_solution TEXT NOT NULL,
  category TEXT NOT NULL,
  potential TEXT NOT NULL,
  likes INTEGER NOT NULL DEFAULT 0,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

ALTER TABLE public.opportunities ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Lectura publica" ON public.opportunities FOR SELECT TO anon, authenticated USING (true);
CREATE POLICY "Insercion publica" ON public.opportunities FOR INSERT TO anon, authenticated WITH CHECK (true);
CREATE POLICY "Update likes publico" ON public.opportunities FOR UPDATE TO anon, authenticated USING (true) WITH CHECK (true);
